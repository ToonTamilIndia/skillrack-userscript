# Id 8504 - Shorter Left Wind-Mills

```c
#include <stdio.h>
#include <stdlib.h>

int main(){
  int n;
  scanf("%d", &n);

  int* heights = (int*)malloc(n * sizeof(int));
  for(int i = 0; i < n; i++){
    scanf("%d", &heights[i]);
  }

  int* stack = (int*)malloc(n * sizeof(int));
  int* result = (int*)malloc(n * sizeof(int));
  int stack_size = 0;

  for(int i = 0; i < n; i++){
    int count = 0;

    while(stack_size > 0 && heights[stack[stack_size - 1]] < heights[i]){
      int idx = stack[stack_size - 1];
      stack_size--;
      count += result[idx] + 1;
    }

    result[i] = count;
    stack[stack_size] = i;
    stack_size++;
  }

  for(int i = 0; i < n; i++){
    if(i > 0) printf(" ");
    printf("%d", result[i]);
  }
  printf("\n");

  free(heights);
  free(stack);
  free(result);

  return 0;
}
```
