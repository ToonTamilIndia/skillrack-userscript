#include <stdio.h>
#include <string.h>

int main() {
    char input[105];
    fgets(input, sizeof(input), stdin);

    int len = strlen(input);
    if (input[len - 1] == '\n') {
        input[len - 1] = '\0';
        len--;
    }

    char stack[105];
    int top = -1;

    for (int i = 0; i < len; i++) {
        if (input[i] == '*') {
            if (top >= 0) {
                top--;
            }
        } else {
            stack[++top] = input[i];
        }
    }

    for (int i = 0; i <= top; i++) {
        printf("%c", stack[i]);
    }
    printf("\n");

    return 0;
}
