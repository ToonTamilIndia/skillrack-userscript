# Id 2530 - Polygon Perimeter

#include <stdio.h>
int main()
{
    int N;
    scanf("%d",&N);
    int perimeter=0, counter=0;
    int side;
    while(++counter <= N)
    {
        scanf("%d",&side);
        perimeter+=side;
    }
    printf("%d",perimeter);
}
