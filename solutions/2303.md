# Id 2303 - Largest among N numbers

```c
#include<stdio.h>

int main()
{
    int N;
    scanf("%d",&N);

    int largest=0;
    int ctr=N;
    int curr;

    while(ctr > 0)
    {
        scanf("%d",&curr);
        if(largest < curr)
        {
            largest = curr;
        }
        ctr--;
    }

    printf("%d",largest);

}
```
