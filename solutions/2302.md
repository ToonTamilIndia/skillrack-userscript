# Id 2302 - Sum of N numbers

```c
#include<stdio.h>

int main()
{
    int N;
    scanf("%d",&N);

    int sum=0;
    int ctr=N;
    int curr;

    while(ctr > 0)
    {
        scanf("%d",&curr);
        sum+=curr;
        ctr--;
    }

    printf("%d",sum);

}
```
