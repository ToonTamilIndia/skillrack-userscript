int findSequence(int SIZE, int *arr) {
    int k, i;
    for (k = 1; k < SIZE; k++) {
        int ok = 1;
        for (i = 0; i + 1 < k; i++)
            if (arr[i] <= arr[i + 1]) { ok = 0; break; }
        for (i = k; i + 1 < SIZE; i++)
            if (arr[i] >= arr[i + 1]) { ok = 0; break; }
        if (ok) return 1;
    }
    return 0;
}
