# Id 2300 - Reverse Number

```c
#include<stdio.h>

int main()
{
    int N;
    scanf("%d",&N);

    int newNum=0;
    while(N > 0)
    {
        newNum = newNum*10;
        newNum += N%10;
        N = N/10;
    }

    printf("%d",newNum);
}
```
